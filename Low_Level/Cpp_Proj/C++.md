```c++ 

void basic_cpp() {
    // Basic input/output
    std::cout << "Hello, World!" << std::endl;

    // Variables and types
    int a = 10;
    double b = 20.5;
    std::string str = "C++17";

    // Control structures
    if (a < b) {
        std::cout << a << " is less than " << b << std::endl;
    } else {
        std::cout << a << " is not less than " << b << std::endl;
    }

    for (int i = 0; i < 5; ++i) {
        std::cout << "Iteration: " << i << std::endl;

}
```
